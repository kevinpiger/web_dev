`timescale 1ns / 1ps
//====================================================================
// 由 WaveDrom JSON 自動產生的波形重現 testbench
// 這是「把波形放回時間軸」，不是 RTL 設計推論。
// 時間基準：1 拍 = BEAT_NS = 10 ns（WaveDrom 只有拍格,
//           沒有絕對時間，BEAT_NS 是人為指定的。）
//
// ★★ 產碼警告 ★★
//   Common Clock - clk: wave 含省略段 '|'，橫軸不是等比例時間軸 —— 省略段兩側的拍距不可直接換算成延遲
//   Cycle - cyc_o: wave 含省略段 '|'，橫軸不是等比例時間軸 —— 省略段兩側的拍距不可直接換算成延遲
//   Address - adr_o: wave 含省略段 '|'，橫軸不是等比例時間軸 —— 省略段兩側的拍距不可直接換算成延遲
//   Data - dat_o: wave 含省略段 '|'，橫軸不是等比例時間軸 —— 省略段兩側的拍距不可直接換算成延遲
//   Write Enable - we_o: wave 含省略段 '|'，橫軸不是等比例時間軸 —— 省略段兩側的拍距不可直接換算成延遲
//   Strobe - stb_o: wave 含省略段 '|'，橫軸不是等比例時間軸 —— 省略段兩側的拍距不可直接換算成延遲
//   Data - dat_i: wave 含省略段 '|'，橫軸不是等比例時間軸 —— 省略段兩側的拍距不可直接換算成延遲
//   Acknowledgment - ack_i: wave 含省略段 '|'，橫軸不是等比例時間軸 —— 省略段兩側的拍距不可直接換算成延遲
//====================================================================

module wave_tb;

  localparam real BEAT_NS = 10.0;

  // 資料匯流排符號：原圖只給標籤，真實編碼未知，
  // 以下為「彼此相異」的佔位值，不代表協定實際編碼。
  localparam [3:0] SYM_VALID = 4'd0;

  reg Common_Clock_clk;
  reg Cycle_cyc_o;
  reg [3:0] Address_adr_o;
  reg Data_dat_o;
  reg Write_Enable_we_o;
  reg Strobe_stb_o;
  reg [3:0] Data_dat_i;
  reg Acknowledgment_ack_i;

  // ---- Common Clock - clk ----
  initial begin
    Common_Clock_clk = 1'b1;
    #(1 * BEAT_NS);
    Common_Clock_clk = 1'b0;
    #(1 * BEAT_NS);
    Common_Clock_clk = 1'b1;
    #(1 * BEAT_NS);
    Common_Clock_clk = 1'b0;
    #(1 * BEAT_NS);
    Common_Clock_clk = 1'b1;
    #(1 * BEAT_NS);
    Common_Clock_clk = 1'b0;
  end

  // ---- Cycle - cyc_o ----
  initial begin
    Cycle_cyc_o = 1'b0;
    #(2 * BEAT_NS);
    Cycle_cyc_o = 1'b1;
    Cycle_cyc_o = 1'b0;
  end

  // ---- Address - adr_o ----
  initial begin
    Address_adr_o = 4'bx;
    #(2 * BEAT_NS);
    Address_adr_o = SYM_VALID;
    Address_adr_o = 4'bx;
  end

  // ---- Data - dat_o ----
  initial begin
    Data_dat_o = 1'bx;
  end

  // ---- Write Enable - we_o ----
  initial begin
    Write_Enable_we_o = 1'b0;
  end

  // ---- Strobe - stb_o ----
  initial begin
    Strobe_stb_o = 1'b0;
    #(2 * BEAT_NS);
    Strobe_stb_o = 1'b1;
    #(2 * BEAT_NS);
    Strobe_stb_o = 1'b0;
  end

  // ---- Data - dat_i ----
  initial begin
    Data_dat_i = 4'bx;
    Data_dat_i = SYM_VALID;
    #(2 * BEAT_NS);
    Data_dat_i = 4'bx;
  end

  // ---- Acknowledgment - ack_i ----
  initial begin
    Acknowledgment_ack_i = 1'b0;
    Acknowledgment_ack_i = 1'b1;
    #(2 * BEAT_NS);
    Acknowledgment_ack_i = 1'b0;
  end

  initial begin
    $dumpfile("wave_tb.vcd");
    $dumpvars(0, wave_tb);
    #(12 * BEAT_NS + BEAT_NS);
    $finish;
  end

endmodule
